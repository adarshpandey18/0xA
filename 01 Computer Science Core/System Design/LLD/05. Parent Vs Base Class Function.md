package org.adarsh.api;  
  
import org.adarsh.boards.TicTacToe;  
import org.adarsh.game.GameResult;  
import org.adarsh.game.Board;  
import org.adarsh.game.Move;  
import org.adarsh.game.Player;  
  
class GameEngine{  
    public Board start() {  
        return new Board();  
    }  
  
    public void move(Board board, Player player, Move move) {  
  
    }  
  
    public GameResult isComplete(Board board) {  
        if(board instanceof TicTacToe) {  
            TicTacToe board1 = (TicTacToe) board;  
            String firstCharacter  = "-";  
            boolean rowComplete = true;  
            for (int i = 0; i < 3; i++) {  
                rowComplete = true;  
                firstCharacter = board1.get(i,0);  
                for (int j = 0; j < 3; j++) {  
                    if(!board1.get(i,j).equals(firstCharacter)) {  
                        rowComplete = false;  
                        break;  
                    }  
                }  
  
                if(rowComplete) {  
                    return  new GameResult(true, firstCharacter);  
                }  
            }  
            boolean colComplete = true;  
            for (int i = 0; i < 3; i++) {  
                colComplete = true;  
                firstCharacter = board1.get(0,i);  
                for (int j = 0; j < 3; j++) {  
                    if(!board1.get(j,i).equals(firstCharacter)) {  
                        colComplete = false;  
                        break;  
                    }  
                }  
  
                if(colComplete) {  
                    return  new GameResult(true, firstCharacter);  
  
                }  
            }  
  
            boolean diagComplete = true;  
            for (int i = 0; i < 3; i++) {  
                diagComplete = true;  
                firstCharacter = board1.get(0,0);  
                if(!board1.get(i,i).equals(firstCharacter)) {  
                    diagComplete = false;  
                    break;  
                }  
            }  
            if(diagComplete) {  
                return  new GameResult(true, firstCharacter);  
  
            }  
            boolean revDiagComplete = true;  
            for (int i = 0; i < 3; i++) {  
                revDiagComplete = true;  
                firstCharacter = board1.get(0,2);  
                if(!board1.get(i,2-1).equals(firstCharacter)) {  
                    revDiagComplete = false;  
                    break;  
                }  
            }  
            if(revDiagComplete) {  
                return  new GameResult(true, firstCharacter);  
            }  
  
            int countOfFilledCells = 0;  
            for (int i = 0; i < 3; i++) {  
                for (int j = 0; j < 3; j++) {  
                    if(board1.get(i, j) != null) {  
                        countOfFilledCells++;  
                    }  
                }  
            }  
  
            if(countOfFilledCells == 12) {  
                return new GameResult(false, "-");  
            }  
        }  
            return new GameResult(false, "No Winner");  
    }  
}

package org.adarsh.boards;

public class TicTacToe extends Board {  
    String[][] cells = new String[3][3];  
  
    public  String get(int row, int col) {  
        return cells[row][col];  
    }  
}
